cmake_minimum_required(VERSION 3.20)
project(fluxledger VERSION 0.1.0 LANGUAGES CXX)
option(FLUXLEDGER_PYTHON "Build Python bindings" ON)
include(CTest)
add_library(transport INTERFACE)
target_include_directories(transport INTERFACE cpp)
target_compile_features(transport INTERFACE cxx_std_17)
if(MSVC)
  target_compile_options(transport INTERFACE /W4 /WX)
else()
  target_compile_options(transport INTERFACE -Wall -Wextra -Wpedantic -Werror)
endif()
if(FLUXLEDGER_PYTHON)
  find_package(Python COMPONENTS Interpreter Development.Module REQUIRED)
  find_package(pybind11 CONFIG REQUIRED)
  pybind11_add_module(_core cpp/bindings.cpp)
  target_link_libraries(_core PRIVATE transport)
  install(TARGETS _core LIBRARY DESTINATION fluxledger RUNTIME DESTINATION fluxledger)
endif()
if(BUILD_TESTING)
  add_executable(test_transport cpp/test_transport.cpp)
  target_link_libraries(test_transport PRIVATE transport)
  add_test(NAME transport_invariants COMMAND test_transport)
endif()
